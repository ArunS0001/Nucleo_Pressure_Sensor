def setup_vscode_config():
    # Verify project target directory exists
    if not os.path.exists(TARGET_DIR):
        print(f"Error: Target directory '{TARGET_DIR}' does not exist.")
        return

    # Create .vscode directory if it doesn't exist
    os.makedirs(VSCODE_DIR, exist_ok=True)

    # Write c_cpp_properties.json
    cpp_path = os.path.join(VSCODE_DIR, "c_cpp_properties.json")
    with open(cpp_path, "w", encoding="utf-8") as f:
        json.dump(c_cpp_props, f, indent=4)
    print(f"Successfully created: {cpp_path}")

    # Write tasks.json
    tasks_path = os.path.join(VSCODE_DIR, "tasks.json")
    with open(tasks_path, "w", encoding="utf-8") as f:
        json.dump(tasks, f, indent=4)
    print(f"Successfully created: {tasks_path}")

if __name__ == "__main__":
    setup_vscode_config()
